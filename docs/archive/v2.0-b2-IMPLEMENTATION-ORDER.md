### 2.2 注入依赖（**修正点 2**）

`agent-ipc.js` 里**没有 `brain`**。在 `:107` 的 `registerAgentIpc(ipcGuard, agentContext, options = {})` 内，与既有 `options.registry || ctx.registry` 同构地取：

```js
  // v2.0-b2：Ticket 生命周期（只读存储 + 主进程填字段）
  //   ⚠️ 刻意**不在这里 new TicketStore()**——那会造出第二个实例，
  //      两个 store 各持一份 tickets 数组：面板建的单子注入时读不到。
  //      唯一实例在 brain.js:305，由 main.js 注入进来。
  const ticketStore = options.ticketStore || ctx.ticketStore || null;
  const getWindowNo = (typeof options.getWindowNo === 'function')
    ? options.getWindowNo
    : (typeof ctx.getWindowNo === 'function' ? ctx.getWindowNo : () => 1);
```

`main.js`（`:2955` 那处 `registerAgentIpc` 调用）的第二组 options 里加：

```js
    ticketStore: brain && brain.ticketStore ? brain.ticketStore : null,
    getWindowNo: () => (brain && brain.memory && brain.memory.data
      && Number.isFinite(brain.memory.data.windowNo)) ? brain.memory.data.windowNo : 1,
```

### 2.3 四条 handler（**核心：错误字段归一 + 只摘白名单字段**）

放在 `agent:get-audit-log`（`:1061`）之后。**每条都要做 `code → error` 归一**（模板见 `:1009-1010`）：

```js
/* v2.0-b2：Ticket 四条通道。
 *
 * 🔴 三条铁律：
 *   1. 错误字段归一到 `error`（TicketStore 内部用 `code`）——渲染端只认一种形状。
 *      参照 agent:execute 的 `const { ok, code, ...rest } = result; return { ok:false, ...rest, error: code }`。
 *   2. 只**显式摘取**白名单字段：`id` / `windowNo` / `now` **一律不接受渲染端传入**。
 *      Ticket 是跨轮状态的注入源，能伪造窗口号 = 能让任意文字出现在她下一轮视野里。
 *      （这不是越权执行，是越权影响认知。）
 *   3. 全部走 guard() + assertByHandler()，与既有 8 条同构。 */
function ticketReply(result) {
  if (!result || result.ok === true) return result;
  const { ok, code, ...rest } = result;
  return { ok: false, ...rest, error: code || 'TICKET_ERROR' };
}

// 9. 列表（只读，含全部状态；面板据此渲染）
ipcGuard.handle('agent:ticket-list', guard((event) => {
  assertByHandler(event);
  if (!ticketStore) return fail('TICKETS_UNAVAILABLE');
  return { ok: true, tickets: ticketStore.list() };
}));

// 10. 建一条（id 与 windowNo 由主进程填）
ipcGuard.handle('agent:ticket-create', guard((event, raw = {}) => {
  assertByHandler(event);
  if (!ticketStore) return fail('TICKETS_UNAVAILABLE');
  const r = ticketStore.createTicket({
    title: raw && raw.title,              // 只摘这三个
    note: raw && raw.note,
    capability: raw && raw.capability,
    windowNo: getWindowNo(),              // ← 主进程填，忽略渲染端
    // ⚠️ 刻意不传 id / now / state：createTicket 内部生成或走默认
  });
  return ticketReply(r);
}));

// 11. 改状态
ipcGuard.handle('agent:ticket-transition', guard((event, raw = {}) => {
  assertByHandler(event);
  if (!ticketStore) return fail('TICKETS_UNAVAILABLE');
  const r = ticketStore.transition(raw && raw.id, raw && raw.state, {
    note: raw && raw.note,
    windowNo: getWindowNo(),              // 主进程填
  });
  return ticketReply(r);
}));

// 12. 改备注
ipcGuard.handle('agent:ticket-note', guard((event, raw = {}) => {
  assertByHandler(event);
  if (!ticketStore) return fail('TICKETS_UNAVAILABLE');
  const r = ticketStore.updateNote(raw && raw.id, raw && raw.note, {
    windowNo: getWindowNo(),
  });
  return ticketReply(r);
}));
```

> ⚠️ `transition` 的第三个参数在 `agent-lifecycle.js:257` 是 `opts`，里面读 `opts.note` / `opts.windowNo` / `opts.now`。**`now` 不给**——让 `TicketStore` 用 `Date.now()`，避免渲染端影响 `updatedAt`（那会改变注入排序）。

### 2.4 三条硬约束（必须有反例测试）

| # | 约束 | 为什么 |
|---|---|---|
| **1** | `windowNo` 由主进程填，渲染端传入的忽略 | 见上 |
| **2** | `id` 由 `createTicket` 内部生成，不接受外部指定 | 伪造 id 可覆盖/冒充已有条目 |
| **3** | 错误字段统一为 `error`，不把 `code` 漏出去 | 渲染端两套字段会「同一错误两种表现」 |

```js
async function refreshTickets() {
  const list = document.getElementById('ticketList');
  if (!list) return;
  const api = window.deskpet && window.deskpet.agent;
```

### 4.1 先说清能做与不能做

若今天只做 A，真机观察有两条临时路径：

- 直接往 `%APPDATA%\Life\agent-tickets.json` 手写一条（注意 JSON 结构：`{schemaVersion, updatedAt, tickets:[...]}`，每条需 `id/title/state/createdAt/updatedAt`）；
- 或用一次性 Node 脚本 `new TicketStore({dir})` 建一条（**不要在 Electron 运行时里跑，会和正在运行的实例抢文件**）。

### 4.2 写进 `docs/PENDING-TASKS.md` 的流程

```
### 验证：她认不认得「未完事项」

前置：建一条 Ticket「整理沙盒里的旧便签」，备注「还差三张」

步骤：
1. 确认 %APPDATA%\Life\agent-tickets.json 已生成且含该条
2. 随便聊一句（触发一次 build）
3. 看她的回复

三种结果，含义完全不同：
  (a) 她提起了这件事      → 注入生效且语义可读。成功。
  (b) 她完全没提          → ⚠️ 可能正常。她读到但选择不说，是合法的。
                          区分方法：临时把头部改成「【测试】你必须提到：XXX」再看。
                          若此时她能提，说明注入没问题，只是她选择不说。
```

> ⚠️ **(b) 是最容易被误判的分支。**
> 「她读到但不说」和「她没读到」从外部看一样，但一个是**她的选择**，一个是**系统的失败**。
> 把前者当后者去「修」，等于要求她必须表演。
